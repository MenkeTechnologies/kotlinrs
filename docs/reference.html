>_LANGUAGE REFERENCE

Every reserved keyword, built-in type, and builtin call the current kotlinrs build recognizes, grouped by keyword then type then builtin. This page is generated from the language-server corpus (src/lsp.rs) by the gen-docs binary, so it stays in sync with what the runtime and editor tooling actually know about. Keywords mirror lexer.rs, types mirror ast::Type::from_name, and each builtin mirrors a real arm in compiler::compile_call.

Keyword

# fun

declare a function; execution enters `fun main`

fun square(n: Int): Int { return n * n }

# val

declare a read-only (immutable) binding

val x = 41
println(x + 1)   // 42

# var

declare a reassignable (mutable) binding

var i = 0
i += 1   // i == 1

# if

conditional branch; also usable as an expression

val m = if (a > b) a else b

# else

the fallback branch of an `if`

if (n % 2 == 0) println("even") else println("odd")

# while

loop while the condition stays true

var i = 0
while (i < 3) { i += 1 }

# for

iterate a range: `for (x in a..b)`

for (i in 1..5) println(i)

# in

the `for (x in iterable)` separator, and the membership operator

for (c in 0 until 3) println(c)
println(2 in 1..3)   // true

# return

return a value (or Unit) from the current function

fun answer(): Int { return 42 }

# until

half-open ascending range: `a until b` excludes b

for (i in 0 until 3) println(i)   // 0 1 2

# ..

inclusive range: `a..b` builds an IntRange value

val r = 1..5
println(r.sum())   // 15

# downTo

descending inclusive range: `a downTo b`

for (i in 3 downTo 1) println(i)   // 3 2 1

# step

stride for a range: `a..b step n`

for (i in 0..10 step 2) println(i)

# true

the Boolean true literal

val ok: Boolean = true

# false

the Boolean false literal

val done: Boolean = false

# when

multi-way branch (statement or expression); subject or subjectless

when (n) { 1 -> "one"; in 2..9 -> "few"; else -> "many" }

# is

runtime type check, in a `when` arm or any expression: `is String`, `!is Int`

when (x) { is String -> "str"; is Int -> "int"; else -> "?" }

# class

declare a class; `: Super(args), Iface` names its supertypes

open class Animal(val name: String)
class Dog(n: String) : Animal(n)

# interface

declare an interface; members may carry a default body

interface Greeter {
    fun greet(): String
    fun loud(): String = greet().uppercase()
}

# object

declare a singleton, built once and reachable by name

object Counter { var n = 0 }
println(Counter.n)

# data

`data class` — structural equals/hashCode/toString plus copy and componentN

data class Pt(val x: Int, val y: Int)
println(Pt(1, 2))   // Pt(x=1, y=2)

# open

mark a class or member as extendable/overridable

open class Base { open fun f(): Int = 1 }

# override

replace a supertype's member; dispatch picks the receiver's runtime class

class D : Base() { override fun f(): Int = 2 }

# abstract

a class that cannot be instantiated, or a member with no body

abstract class Shape { abstract fun area(): Double }

# sealed

an abstract class whose subtypes make a `when` over `is` arms exhaustive

sealed class Expr
data class Num(val v: Int) : Expr()

# super

call the supertype's implementation instead of the overriding one

override fun f(): Int = super.f() + 1

# break

exit the enclosing loop; `break@label` targets a labeled loop

for (i in 1..9) { if (i == 5) break }

# continue

skip to the loop's next iteration; `continue@label` for a labeled loop

for (i in 1..9) { if (i % 2 == 0) continue; println(i) }

# try

guarded block; an EXPRESSION whose value is the body's or the matching handler's

val n = try { "x".length } catch (e: Exception) { -1 }

# catch

handler arm: `catch (e: T) { … }`; the first arm whose type matches wins

try { 1 / 0 } catch (e: ArithmeticException) { println(e.message) }

# finally

cleanup block; runs on the normal and the exceptional path alike

try { work() } finally { println("done") }

# throw

raise a throwable; typed `Nothing`, so it is usable as an expression

val v = x ?: throw IllegalStateException("missing")

# null

the null reference; used with `T?`, `?.`, `?:`, and `!!`

val x: Int? = null
println(x ?: 0)   // 0

Type

# Int

32/64-bit signed integer; `/` and `%` truncate toward zero

val n: Int = 7 / 2   // 3

# Long

64-bit signed integer; shares Int's integer division rules

val big: Long = 1000

# Double

IEEE-754 double; prints with a trailing `.0` when whole

val d: Double = 3.0   // "3.0"

# Float

floating-point value (coerced to Double behavior here)

val f: Float = 1.5f

# Boolean

true/false; prints as `true` or `false`

val b: Boolean = 1 < 2

# Char

a single character (code unit); integral — `'A' + 1`, `.code`, `.toChar()`

val c: Char = 'A'
println(c + 1)   // B

# String

text; `+` concatenates and `"$x"` interpolates

val s: String = "n = ${1 + 1}"

# Unit

the no-value type; the result of a function with no return

fun log(): Unit { println("hi") }

Builtin

# println

write a value to stdout followed by a newline

println(6 * 7)   // 42

# print

write a value to stdout with no trailing newline

print("a"); print("b")   // ab

# listOf

build a read-only List; `mutableListOf` builds a mutable one

val xs = listOf(1, 2, 3)
println(xs.size)   // 3

# mapOf

build a Map from `k to v` pairs; `mutableMapOf` builds a mutable one

val m = mapOf("a" to 1)
println(m["a"])   // 1

# arrayOf

build an array; `intArrayOf`/`doubleArrayOf`/`booleanArrayOf` are typed

val a = arrayOf(1, 2, 3)
println(a[1])   // 2

# IntArray

a zero-filled primitive array; also `DoubleArray`/`BooleanArray`

val a = IntArray(3)
println(a.sum())   // 0

# abs

absolute value (kotlin.math; needs `import kotlin.math.*`)

import kotlin.math.*
println(abs(-3))   // 3

# sqrt

square root of a Double (kotlin.math; needs the import)

import kotlin.math.*
println(sqrt(9.0))   // 3.0

# max

larger of two values (kotlin.math); `maxOf` needs no import

println(maxOf(2, 9))   // 9

# min

smaller of two values (kotlin.math); `minOf` needs no import

println(minOf(2, 9))   // 2

# Math

the java.lang.Math statics, auto-imported (abs/max/min/sqrt/round/…)

println(Math.abs(-3))   // 3
